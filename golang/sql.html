连接池

sql.DB 是连接池，不是单个连接

核心

sql.DB 是一个连接池管理器，内部维护一组数据库连接，并发安全。每次执行查询时自动从池中借用连接，用完归还，不需要手动管理连接生命周期。

db.Ping() 用于验证连接可用性（会从池中取一个连接发送 ping 命令）。db.Stats() 返回连接池当前状态统计。

import (
    "database/sql"
    _ "github.com/go-sql-driver/mysql"  // 注册驱动（副作用导入）
)

func initDB() *sql.DB {
    db, err := sql.Open("mysql", "user:pass@tcp(127.0.0.1:3306)/dbname?parseTime=true")
    if err != nil { panic(err) }

    // 连接池调优参数
    db.SetMaxOpenConns(25)                   // 最大打开连接数（含使用中+空闲）
    db.SetMaxIdleConns(10)                   // 最大空闲连接数
    db.SetConnMaxLifetime(5 * time.Minute)  // 连接最大存活时间（防止 MySQL 超时断开）
    db.SetConnMaxIdleTime(2 * time.Minute)  // 连接最大空闲时间（Go 1.15+）

    if err = db.Ping(); err != nil { panic(err) }
    return db
}

连接池参数调优策略

进阶
参数作用推荐值
SetMaxOpenConns(n)限制并发连接总数，防止数据库过载数据库最大连接数 / 服务实例数
SetMaxIdleConns(n)保持空闲连接避免频繁建立MaxOpenConns 的 30~50%
SetConnMaxLifetime(d)强制连接定期重建，防止 MySQL 8h 超时小于 MySQL wait_timeout（默认 8h），建议 5min
SetConnMaxIdleTime(d)空闲连接超时回收，释放数据库资源1~5 分钟

注意：SetMaxIdleConns 的值不应超过 SetMaxOpenConns，否则多余的空闲连接限制无效。若 MaxIdleConns=0，每次用完连接立即关闭，没有连接复用。

查询与 Scan

db.Query vs db.QueryRow

核心

db.Query 返回 *sql.Rows，用于多行结果；db.QueryRow 返回 *sql.Row，用于期望单行结果（找不到时 Scan 返回 sql.ErrNoRows）。

// 多行查询 — 必须调用 rows.Close()
func getUsers(db *sql.DB) ([]User, error) {
    rows, err := db.Query("SELECT id, name, email FROM users WHERE active = ?", true)
    if err != nil { return nil, err }
    defer rows.Close()  // 关键！不关闭则连接不归还连接池

    var users []User
    for rows.Next() {
        var u User
        // Scan 参数顺序必须与 SELECT 列顺序一致
        if err := rows.Scan(&u.ID, &u.Name, &u.Email); err != nil {
            return nil, err
        }
        users = append(users, u)
    }
    return users, rows.Err()  // 检查迭代过程中的错误
}

// 单行查询
func getUserByID(db *sql.DB, id int) (User, error) {
    var u User
    err := db.QueryRow("SELECT id, name, email FROM users WHERE id = ?", id).
        Scan(&u.ID, &u.Name, &u.Email)
    if err == sql.ErrNoRows { return User{}, nil }
    return u, err
}

sql.NullString 处理 NULL 值

核心

数据库列可能为 NULL，直接 Scan 到 string 会报错。使用 sql.NullString、sql.NullInt64、sql.NullTime 等 Null 类型安全处理。

type Profile struct {
    ID     int
    Name   string
    Phone  sql.NullString   // 可为 NULL
    Score  sql.NullFloat64  // 可为 NULL
    BornAt sql.NullTime     // 可为 NULL（Go 1.15+）
}

var p Profile
rows.Scan(&p.ID, &p.Name, &p.Phone, &p.Score, &p.BornAt)

// 使用时需检查 Valid 字段
if p.Phone.Valid {
    fmt.Println(p.Phone.String)  // 实际电话号码
} else {
    fmt.Println("no phone")      // 数据库为 NULL
}

事务

defer rollback 惯用法

核心

事务的正确写法：先 defer tx.Rollback()，成功时再 tx.Commit()。Commit 成功后 Rollback 是空操作（error 被忽略），这样无论函数以何种方式退出都能保证回滚。

func transfer(db *sql.DB, fromID, toID, amount int) error {
    tx, err := db.Begin()
    if err != nil { return err }
    defer tx.Rollback()  // commit 后 rollback 是 no-op，安全

    // 操作1：扣款
    _, err = tx.Exec("UPDATE accounts SET balance = balance - ? WHERE id = ?", amount, fromID)
    if err != nil { return err }  // defer 触发 Rollback

    // 操作2：入账
    _, err = tx.Exec("UPDATE accounts SET balance = balance + ? WHERE id = ?", amount, toID)
    if err != nil { return err }

    // 验证余额不为负
    var bal int
    tx.QueryRow("SELECT balance FROM accounts WHERE id = ?", fromID).Scan(&bal)
    if bal < 0 { return errors.New("insufficient balance") }

    return tx.Commit()  // 成功提交，defer Rollback 成为 no-op
}

BeginTx 与事务隔离级别

进阶

db.BeginTx(ctx, opts) 支持 Context 取消和自定义隔离级别。推荐在服务层传入 Context，使请求取消时事务能及时回滚。

func transferWithCtx(ctx context.Context, db *sql.DB) error {
    tx, err := db.BeginTx(ctx, &sql.TxOptions{
        Isolation: sql.LevelSerializable,  // 最高隔离级别
        ReadOnly:  false,
    })
    if err != nil { return err }
    defer tx.Rollback()
    // ... 执行事务操作 ...
    return tx.Commit()
}

// 隔离级别常量
// sql.LevelDefault         — 驱动默认（通常是 Read Committed）
// sql.LevelReadUncommitted — 读未提交（最低，有脏读）
// sql.LevelReadCommitted   — 读已提交（避免脏读）
// sql.LevelRepeatableRead  — 可重复读（MySQL 默认，避免不可重复读）
// sql.LevelSerializable    — 串行化（最高，避免幻读，性能最低）

Prepared Statement

db.Prepare 与防 SQL 注入

核心

Prepared Statement 将 SQL 模板预先发送到数据库编译，执行时只传参数。优势：防 SQL 注入（参数不参与 SQL 解析）、多次执行时减少解析开销。

所有带用户输入的 SQL 都应使用占位符 ?，而非字符串拼接！

// 错误：SQL 注入风险
query := fmt.Sprintf("SELECT * FROM users WHERE name = '%s'", userInput)
// 输入 "' OR '1'='1" 会返回全表数据！

// 正确：使用占位符参数
rows, err := db.Query("SELECT * FROM users WHERE name = ?", userInput)

// Prepare：批量执行同一 SQL 时更高效
func batchInsert(db *sql.DB, users []User) error {
    stmt, err := db.Prepare("INSERT INTO users(name, email) VALUES(?, ?)")
    if err != nil { return err }
    defer stmt.Close()  // 用完必须关闭，释放数据库端游标

    for _, u := range users {
        if _, err = stmt.Exec(u.Name, u.Email); err != nil {
            return err
        }
    }
    return nil
}

Prepare 的代价与适用场景

进阶

db.Prepare 实际上要额外发送一次网络请求（发送模板、获取 statement ID）。对于只执行一次的 SQL，db.Exec/Query 直接传参即可，驱动内部仍会处理参数转义，无注入风险，且少一次网络往返。

方式网络往返适用场景
db.Exec("... ?", args)1次（直接执行）单次执行，简洁首选
db.Prepare + stmt.Exec2次（prepare + exec）同一 SQL 循环执行多次
tx.Stmt(stmt)0次额外（复用已有 stmt）事务中复用连接级 stmt
// 事务中使用 Prepare：在 tx 上 Prepare，确保 stmt 使用同一连接
tx, _ := db.Begin()
defer tx.Rollback()

stmt, _ := tx.Prepare("UPDATE inventory SET qty = qty - ? WHERE sku = ?")
defer stmt.Close()

for _, item := range orderItems {
    stmt.Exec(item.Qty, item.SKU)
}
tx.Commit()

面试题

sql.DB 是否是线程安全的？每次调用都会创建新连接吗？▶
线程安全：sql.DB 是并发安全的，可以被多个 goroutine 同时使用，不需要额外加锁。

不是每次创建新连接：sql.DB 是连接池，内部维护一个空闲连接队列。执行查询时从池中取可用连接，用完后归还。只有当池中没有空闲连接且当前总连接数 < MaxOpenConns 时，才会创建新连接。

sql.Open 本身并不建立连接，只是初始化池结构。第一次实际查询时才建立连接。用 db.Ping() 可以提前触发连接建立并验证可用性。
如何正确处理查询结果集？rows.Close() 什么时候调用？▶
标准处理流程：
1. rows, err := db.Query(...)，先检查 err
2. 立即 defer rows.Close()（确保任何退出路径都会关闭）
3. for rows.Next() 迭代，每次调用 rows.Scan(...)
4. 循环结束后检查 rows.Err()（迭代中途的错误）

为什么必须 Close？
rows 持有一个数据库连接，不 Close 则连接不归还连接池，最终耗尽连接导致所有查询挂起。

注意：rows.Next() 返回 false（遍历完毕）时会自动调用 Close，但如果中途 return/break，必须依赖 defer 来关闭。
如何用 defer 正确写事务的回滚逻辑？▶
标准惯用法：

tx, err := db.Begin()
if err != nil { return err }
defer tx.Rollback() // ← 关键，放在 Begin 后紧跟

之后正常执行业务逻辑，最后调用 tx.Commit()。

工作原理：Commit 成功后，事务已结束，随后 defer 触发的 Rollback 是一个空操作（返回 sql.ErrTxDone，但通常被忽略）。
如果中途 return 了 error，或发生 panic，defer 的 Rollback 会回滚所有操作，保证数据一致性。

注意不要写成 defer func(){ if err != nil { tx.Rollback() }} 这种形式——闭包捕获的 err 在 defer 执行时已被 Commit 改写，逻辑会出错。
Prepared Statement 的优势是什么？什么时候不值得用？▶
优势：
1. 防 SQL 注入：参数独立传输，不与 SQL 文本混合，数据库层面保证安全
2. 减少解析开销：SQL 只编译一次，多次执行复用解析结果（批量操作时有明显收益）
3. 类型安全：参数按类型传输，避免手动转义带来的错误

不值得用的场景：
• 只执行一次的 SQL：Prepare 需要额外一次网络往返（发模板 + 获取 stmt ID），总共两次比直接执行多一次
• 动态条件查询（如不同的 WHERE 子句组合）：每种组合都要 Prepare，管理复杂
• 简单单次 Exec/Query：直接使用占位符参数同样安全，且更简洁
SetMaxOpenConns 和 SetMaxIdleConns 分别影响什么？如何调优？▶
SetMaxOpenConns(n)：限制连接池同时持有的最大连接数（使用中 + 空闲）。超过此限制的请求会阻塞等待连接释放。设置过小导致吞吐量受限，设置过大导致数据库连接数超限（MySQL 默认 max_connections=151）。

SetMaxIdleConns(n)：池中保留的最大空闲连接数。空闲连接可被下次请求直接复用，避免频繁 TCP 握手。设置为 0 相当于禁用连接复用。

调优建议：
• MaxOpenConns = 数据库 max_connections / 服务实例数 × 0.8（留余量）
• MaxIdleConns = MaxOpenConns × 0.3~0.5（高并发服务适当调大）
• MaxIdleConns 不应超过 MaxOpenConns，否则超出部分无效
• ConnMaxLifetime 建议 < MySQL wait_timeout（默认8h），通常设 5min，防止使用已被服务端关闭的连接

易错点

❌ 忘记 rows.Close()，连接泄漏，最终耗尽连接池
func getUsers(db *sql.DB) ([]User, error) {
    rows, err := db.Query("SELECT * FROM users")
    if err != nil { return nil, err }
    // 忘记 defer rows.Close()!

    var users []User
    for rows.Next() {
        var u User
        if err := rows.Scan(&u); err != nil {
            return nil, err // 泄漏！连接未归还
        }
        users = append(users, u)
    }
    return users, nil
}

中途 return 或 panic 时连接未归还，连接池耗尽后所有查询永久挂起。

func getUsers(db *sql.DB) ([]User, error) {
    rows, err := db.Query("SELECT id, name FROM users")
    if err != nil { return nil, err }
    defer rows.Close()  // ✓ 紧跟 Query 后

    var users []User
    for rows.Next() {
        var u User
        if err := rows.Scan(&u.ID, &u.Name); err != nil {
            return nil, err  // defer 确保关闭
        }
        users = append(users, u)
    }
    return users, rows.Err()
}

defer rows.Close() 紧跟 Query 调用，无论如何退出都能归还连接。

❌ 在事务中使用 db.Query 而不是 tx.Query，操作不在事务内
tx, _ := db.Begin()
defer tx.Rollback()

// 扣款（在事务内）
tx.Exec("UPDATE accounts SET bal=bal-100 WHERE id=1")

// 错误：用 db.Query，不在事务内！
// 它用了另一个连接，看不到事务中的变更
rows, _ := db.Query("SELECT bal FROM accounts WHERE id=1")

tx.Commit()

db.Query 获取的是连接池中的另一个连接，读不到当前事务的未提交变更。

tx, _ := db.Begin()
defer tx.Rollback()

// 扣款（在事务内）
tx.Exec("UPDATE accounts SET bal=bal-100 WHERE id=1")

// 正确：用 tx.Query，与 Exec 使用同一连接
rows, _ := tx.Query("SELECT bal FROM accounts WHERE id=1")  // ✓
defer rows.Close()

tx.Commit()

事务内所有操作必须使用 tx（同一连接），确保读写在同一事务上下文中。

❌ Scan 参数与 SELECT 列顺序不一致导致数据错乱
type User struct {
    ID    int
    Name  string
    Email string
}

rows, _ := db.Query(
    "SELECT name, id, email FROM users")
// SELECT 顺序：name, id, email

var u User
rows.Scan(&u.ID, &u.Name, &u.Email)
// 错误！ID 接收了 name（字符串→int 失败或乱值）

Scan 参数按 SELECT 列的返回顺序匹配，与 struct 字段定义顺序无关。

rows, _ := db.Query(
    "SELECT id, name, email FROM users")
// SELECT 顺序：id, name, email

var u User
rows.Scan(&u.ID, &u.Name, &u.Email)  // ✓

// 最佳实践：使用 sqlx 库的 StructScan
// 或按列名显式 Scan，避免顺序问题
// var id int; var name, email string
// rows.Scan(&id, &name, &email)

确保 SELECT 列顺序与 Scan 参数严格对应。推荐使用 sqlx 库的 StructScan 按列名映射。
