strings 包高频 API

包含与前缀/后缀检测

core

最常用的布尔判断函数，底层均基于内存比较，时间复杂度 O(n)。

import "strings"

strings.Contains("seafood", "foo")      // true
strings.ContainsAny("team", "i")       // false
strings.ContainsRune("hello", 'e')     // true

strings.HasPrefix("Gopher", "Go")      // true
strings.HasSuffix("Amigo", "go")       // false  (大小写敏感)

// 大小写不敏感比较
strings.EqualFold("Go", "go")          // true

Split / Join

core

Split 返回 []string，始终非 nil，空串时返回含一个空元素的切片。SplitN 限制分割次数，SplitAfter 保留分隔符。

parts := strings.Split("a,b,c", ",")    // ["a","b","c"]
parts  = strings.Split("", ",")           // [""]  — 注意：长度为1
fields := strings.Fields("  foo bar  baz  ") // ["foo","bar","baz"]

s := strings.Join([]string{"a","b","c"}, "-") // "a-b-c"

// SplitN: 最多分割 n 段
p := strings.SplitN("a:b:c:d", ":", 3) // ["a","b","c:d"]

Replace / Trim / Index / Count

core 
// Replace(s, old, new, n)  n=-1 表示全部替换
strings.Replace("oink oink oink", "oink", "moo", 2)  // "moo moo oink"
strings.ReplaceAll("oink oink", "oink", "moo")        // "moo moo"

// Trim 系列
strings.TrimSpace("  hello  \n")       // "hello"
strings.Trim("¡¡¡hello!!!", "!¡")     // "hello"  (去除字符集)
strings.TrimPrefix("Goodbye", "Good") // "bye"
strings.TrimSuffix("Hello!", "!")     // "Hello"

// Index 系列（返回字节位置，找不到返回 -1）
strings.Index("hello", "ll")           // 2
strings.LastIndex("go gopher", "go")   // 3
strings.IndexRune("hello", 'e')        // 1

// Count
strings.Count("cheese", "e")           // 3
strings.Count("five", "")              // 5  (UTF-8 字符数 + 1)

strings.Builder 与字符串拼接

拼接方式性能对比

perf
方式 适用场景 内存分配 推荐度
s += x少量（<5次）每次分配简单场景
strings.Builder循环 / 大量拼接指数增长，少次★★★★★
bytes.Buffer需要 io.Writer同 Builder★★★★
fmt.Sprintf格式化输出reflect 开销大非热路径
strings.Join已有切片一次分配★★★★★

核心原因：string 是不可变的，+= 每次都会分配新内存并复制，N 次拼接总复制量为 O(n²)。

strings.Builder 正确使用

core 
var b strings.Builder

// 提前预估容量，避免扩容
b.Grow(64)

for i := 0; i < 10; i++ {
    b.WriteString("hello")
    b.WriteByte(' ')
    b.WriteRune('🚀')    // 写入 Unicode 字符
}
result := b.String()   // 零拷贝返回内部 buf 的字符串视图
b.Reset()             // 重置为空；当前实现释放对 buf 的引用

// 注意：Builder 不能复制后使用（内部含指针检测）
// b2 := b  // 运行时 panic: strings: illegal use of non-zero Builder copied by value

fmt.Sprintf 开销分析

high

fmt.Sprintf 使用反射解析格式化动词，即使是简单的 %d 也比 strconv.Itoa 慢 3-5 倍，在热路径中应避免。

// 慢：fmt.Sprintf 用 reflect，每次分配
s1 := fmt.Sprintf("%d", 42)

// 快：strconv，无反射
s2 := strconv.Itoa(42)

// 更快：AppendInt 追加到已有 buf，零分配
var buf []byte
buf = strconv.AppendInt(buf, 42, 10)

bytes 包

bytes.Buffer 与 bytes.Reader

core

bytes.Buffer 实现了 io.Reader、io.Writer、io.ByteWriter 等接口，适合需要传递 Writer 接口的场景（如 json.NewEncoder）。strings.Builder 只实现 Write 方向，不可读。

// bytes.Buffer — 可读可写
var buf bytes.Buffer
buf.WriteString("hello")
buf.WriteByte(' ')
fmt.Fprintf(&buf, "world %d", 42)
s := buf.String()   // 会复制一次
b := buf.Bytes()    // 返回底层 slice，无复制

// bytes.Reader — 只读，零拷贝包装 []byte
r := bytes.NewReader([]byte("hello"))
r.Seek(0, io.SeekStart)  // 支持随机访问

[]byte 与 string 转换开销

high

通常情况下 []byte(str) 和 string(b) 都会产生一次内存拷贝。但 Go 编译器在以下特定模式中会优化为零拷贝：

// 以下场景编译器会优化（不分配）
m[string(b)]          // map 查找
s == string(b)          // 比较
switch string(b) { ... } // switch

// unsafe 包零拷贝转换（生产慎用）
import "unsafe"

func bytesToString(b []byte) string {
    return *(*string)(unsafe.Pointer(&b))
}

// Go 1.20+ 官方提供
s := unsafe.String(&b[0], len(b))  // 零拷贝，b 不能 GC
b2 := unsafe.Slice(unsafe.StringData(s), len(s))

strconv 与编码

Itoa / Atoi 与 ParseInt / FormatInt

core 
// int → string
s := strconv.Itoa(42)                    // "42"，等价 FormatInt(42,10)
s  = strconv.FormatInt(255, 16)           // "ff" (16进制)
s  = strconv.FormatInt(8, 2)             // "1000" (二进制)

// string → int
n, err := strconv.Atoi("42")             // 返回 int
n64, err := strconv.ParseInt("ff", 16, 64) // 基数16，位宽64
_, ok := err.(*strconv.NumError)           // 类型断言检查错误

// bool
b, _  := strconv.ParseBool("true")        // 支持: 1,t,T,true,TRUE
s     = strconv.FormatBool(true)           // "true"

// float
f, _  := strconv.ParseFloat("3.14", 64)
s     = strconv.FormatFloat(f, 'f', 2, 64) // "3.14"

AppendInt 避免分配

perf

Append 系列函数将结果追加到已有 []byte 尾部，避免 FormatInt + 字符串拼接的两次分配，在序列化场景中非常常见。

var buf []byte
buf = strconv.AppendInt(buf, 42, 10)    // buf = "42"
buf = append(buf, ',')
buf = strconv.AppendFloat(buf, 3.14, 'f', 2, 64)
buf = strconv.AppendBool(buf, true)

// Quote / Unquote — 处理字符串转义
strconv.Quote(`He said "hi"`)             // "\"He said \\\"hi\\\"\""
strconv.Unquote(`"hello\nworld"`)         // "hello\nworld"

rune / byte 区别与 utf8 包

core

byte 是 uint8 别名，表示单个字节。rune 是 int32 别名，表示单个 Unicode 代码点（Code Point）。中文字符在 UTF-8 中占 3 字节，所以 len("中文") 返回 6，而 len([]rune("中文")) 返回 2。

s := "Hello, 世界"
fmt.Println(len(s))            // 13 (字节数)
fmt.Println(len([]rune(s)))  // 9  (字符数，会分配)

// utf8 包：不转 []rune，零分配计数字符
utf8.RuneCountInString(s)     // 9
utf8.ValidString(s)           // true

// range 迭代按 rune
for i, r := range s {
    // i 是字节偏移，r 是 rune 值
    fmt.Printf("%d: %c\n", i, r)
}

// 解码单个 rune
r, size := utf8.DecodeRuneInString(s[7:])  // '世', 3

面试题

strings.Builder 比直接用 + 拼接快在哪里？什么时候用 +？ ▶
string 是不可变类型，s += x 每次都分配新内存并把原字符串全部复制过去，N 次拼接总复制量 O(n²)。strings.Builder 内部维护 []byte，按指数增长容量，均摊复制量 O(n)，在循环中拼接时快数十倍甚至数百倍。

使用 + 的合理场景：拼接次数极少（2-3次）、不在循环里、代码可读性更重要。编译器对 "a" + "b" + "c" 这类常量拼接会在编译期完成，不产生运行时开销。
[]byte 和 string 相互转换有没有内存分配？如何优化？ ▶
通常有拷贝。string(b) 分配新字符串内存，[]byte(s) 分配新切片内存。编译器在有限场景（map 查找、比较、switch）下会优化为零拷贝。

优化手段：①减少转换次数，保持同一类型操作；②Go 1.20+ 使用 unsafe.String/unsafe.Slice；③使用 bytes 包代替 strings 包直接操作 []byte，避免中间转换。
strings.Split("", ",") 返回什么？ ▶
返回 []string{""}，即长度为 1、包含一个空字符串的切片，而不是空切片 []string{}。这是一个常见陷阱。如果你需要判断输入是否为空，应先检查 s == ""，再决定是否调用 Split。

对比：strings.Fields("") 返回 []string{}（空切片）。
strconv.Atoi 和 fmt.Sscanf 哪个更快？为什么？ ▶
strconv.Atoi 快 5-10 倍。fmt.Sscanf 内部使用反射解析格式字符串，会产生接口装箱（interface boxing）和额外内存分配；而 strconv.Atoi 是纯算术运算，无反射，无分配。

基准示例（参考）：strconv.Atoi 约 8 ns/op，fmt.Sscanf 约 100 ns/op。在高频解析场景（如日志解析、协议解析）中差异显著。
如何高效地在循环中构建一个字符串？ ▶
首选 strings.Builder：① 调用 b.Grow(estimatedSize) 预分配；② 循环内调用 WriteString/WriteByte；③ 最后调用 b.String()。

如果已知所有片段，直接用 strings.Join(parts, sep)——内部精确计算总长度，只分配一次内存，比 Builder 更快。 避免在循环中使用 fmt.Sprintf 或 +=。

易错点

✗ 循环中用 += 拼接字符串，O(n²) 复制
// 错误：每次都复制
var result string
for _, item := range items {
    result += item  // O(n²)
}

items 有 1000 个元素时，总共复制约 500,000 次字符，性能极差。

// 正确：Builder 均摊 O(n)
var b strings.Builder
b.Grow(len(items) * 10)
for _, item := range items {
    b.WriteString(item)
}
result := b.String()

Grow 预分配后内循环几乎零分配，速度快 10-100 倍。

✗ strings.Builder 不是并发安全的
// 错误：多 goroutine 共享 Builder
var b strings.Builder
var wg sync.WaitGroup
for i := 0; i < 10; i++ {
    wg.Add(1)
    go func() {
        defer wg.Done()
        b.WriteString("x") // 数据竞争!
    }()
}
// 正确：加锁保护
var mu sync.Mutex
var b  strings.Builder
for i := 0; i < 10; i++ {
    wg.Add(1)
    go func() {
        defer wg.Done()
        mu.Lock()
        b.WriteString("x")
        mu.Unlock()
    }()
}

或各 goroutine 用独立 Builder，最后合并。

✗ []byte(str) 转换会复制，除非编译器优化
// 误以为零拷贝
s := "hello world"
b := []byte(s)   // 分配新内存并复制
b[0] = 'H'
// s 不会改变（设计如此）
// 但循环里频繁转换浪费内存

在热路径里反复 []byte(s) 会给 GC 造成大量短生命周期垃圾。

// 保持 []byte 不转换
b := []byte("hello world")
// 直接用 bytes 包操作
b = bytes.ToUpper(b)
b = bytes.TrimSpace(b)
// 只在需要时转一次 string
result := string(b)

在整个处理链中保持 []byte，只在边界处做一次转换。
