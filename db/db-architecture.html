3
查询优化器 Optimizer

1// 数据库核心处理流程
2type DB struct {
3 parser *Parser
4 optimizer *Optimizer
5 executor *Executor
6 txnMgr *TxnManager
7 bufPool *BufferPool
8 disk *DiskManager
9}
10
11func (db *DB) Query(sql string) ([]Row, error) {
12 // Layer 1-2: 解析
13 ast, err := db.parser.Parse(sql)
14 if err != nil {
15 return nil, err
16 }
17 // Layer 3: 优化
18 plan := db.optimizer.Optimize(ast)
19 // Layer 4-5: 执行（含事务）
20 txn := db.txnMgr.Begin()
21 rows, err := db.executor.Execute(plan, txn)
22 if err != nil {
23 txn.Rollback()
24 return nil, err
25 }
26 txn.Commit()
27 return rows, nil
28}

30// Layer 6: 缓冲池页面读取
31func (bp *BufferPool) GetPage(pageID uint32) *Page {
32 if page, ok := bp.cache[pageID]; ok {
33 bp.lru.MoveToFront(page) // O(1) LRU更新
34 return page.data
35 }
36 // Layer 7: 缓存未命中，读磁盘
37 data := bp.disk.ReadPage(pageID)
38 bp.loadPage(pageID, data)
39 return data
40}
