递推公式
// i: A 的下标，j: B 的下标 if A[i-1] == B[j-1]: dp[i][j] = dp[i-1][j-1] + 1 // 字符匹配 else: dp[i][j] = max( dp[i-1][j], // 跳过 A[i-1] dp[i][j-1] // 跳过 B[j-1] )
Go 实现
package main import "fmt" func lcs(a, b string) (int, string) { m, n := len(a), len(b) dp := make([][]int, m+1) for i := range dp { dp[i] = make([]int, n+1) } for i := 1; i <= m; i++ { for j := 1; j <= n; j++ { if a[i-1] == b[j-1] { dp[i][j] = dp[i-1][j-1] + 1 } else if dp[i-1][j] > dp[i][j-1] { dp[i][j] = dp[i-1][j] } else { dp[i][j] = dp[i][j-1] } } } // 回溯 LCS result := []byte{} i, j := m, n for i > 0 && j > 0 { if a[i-1] == b[j-1] { result = append([]byte{a[i-1]}, result...) i--; j-- } else if dp[i-1][j] > dp[i][j-1] { i-- } else { j-- } } return dp[m][n], string(result) } func main() { length, seq := lcs("ABCBDAB", "BDCAB") fmt.Println(length, seq) // 输出: 4 BCAB }
算法说明

LCS：两个序列中都出现、且相对顺序一致的最长子序列（不要求连续）。

dp[i][j] = A 的前 i 个字符与 B 的前 j 个字符的 LCS 长度。

回溯：从 dp[m][n] 出发，字符相等则向左上对角线走，否则走向更大值的方向。

应用：diff 工具、版本控制、基因序列比对、编辑距离。
