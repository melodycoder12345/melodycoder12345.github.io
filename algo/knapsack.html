// i: 物品编号，j: 当前容量 if j < w[i]: dp[i][j] = dp[i-1][j] // 放不下，不选 else: dp[i][j] = max( dp[i-1][j], // 不选第 i 件 dp[i-1][j-w[i]] + v[i] // 选第 i 件 )
Go 实现
package main import "fmt" func knapsack(weights, values []int, W int) int { n := len(weights) dp := make([][]int, n+1) for i := range dp { dp[i] = make([]int, W+1) } for i := 1; i <= n; i++ { w, v := weights[i-1], values[i-1] for j := 0; j <= W; j++ { dp[i][j] = dp[i-1][j] // 不选第 i 件 if j >= w && dp[i-1][j-w]+v > dp[i][j] { dp[i][j] = dp[i-1][j-w] + v } } } return dp[n][W] } func main() { weights := []int{2, 3, 4, 5} values := []int{3, 4, 5, 6} fmt.Println(knapsack(weights, values, 8)) // 输出: 10 }
算法说明

0/1 背包：每件物品只能选或不选，不可拆分。

dp[i][j] = 前 i 件物品、容量为 j 时的最大价值。

初始化：dp[0][*] = 0，dp[*][0] = 0。

回溯路径：从 dp[n][W] 出发，若 dp[i][j] != dp[i-1][j] 则选了物品 i。

优化：空间可压缩到 O(W)（滚动数组，逆序遍历 j）。
