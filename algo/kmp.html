package main
import "fmt"

func buildNext(pat string) []int {
    m := len(pat)
    next := make([]int, m)
    next[0] = 0
    k := 0
    for i := 1; i < m; i++ {
        for k > 0 && pat[k] != pat[i] {
            k = next[k-1]
        }
        if pat[k] == pat[i] { k++ }
        next[i] = k
    }
    return next
}

func kmpSearch(text, pat string) []int {
    next := buildNext(pat)
    var matches []int
    j := 0
    for i := 0; i < len(text); i++ {
        for j > 0 && text[i] != pat[j] {
            j = next[j-1]  // 利用 next 数组跳转
        }
        if text[i] == pat[j] { j++ }
        if j == len(pat) {
            matches = append(matches, i-j+1)
            j = next[j-1]
        }
    }
    return matches
}

func main() {
    text := "ABABDABACDABABCABAB"
    pat  := "ABABCABAB"
    fmt.Println(kmpSearch(text, pat))
}
