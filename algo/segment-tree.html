Segment Tree 线段树

区间求和查询 + 单点更新  |  O(log n) 查询与更新，O(n) 建树
查询/更新：O(log n)
建树：O(n)
空间：O(n)

package main
import "fmt"

type SegTree struct {
    tree []int
    n    int
}

func Build(arr []int) *SegTree {
    n := len(arr)
    tree := make([]int, 4*n)
    var build func(node, l, r int)
    build = func(node, l, r int) {
        if l == r { tree[node] = arr[l]; return }
        mid := (l + r) / 2
        build(2*node, l, mid)
        build(2*node+1, mid+1, r)
        tree[node] = tree[2*node] + tree[2*node+1]
    }
    build(1, 0, n-1)
    return &SegTree{tree, n}
}

func (st *SegTree) Query(node, l, r, ql, qr int) int {
    if ql <= l && r <= qr { return st.tree[node] }  // 完全覆盖
    if qr < l || r < ql  { return 0 }               // 不相交
    mid := (l + r) / 2
    return st.Query(2*node, l, mid, ql, qr) +
           st.Query(2*node+1, mid+1, r, ql, qr)
}

func (st *SegTree) Update(node, l, r, idx, val int) {
    if l == r { st.tree[node] = val; return }
    mid := (l + r) / 2
    if idx <= mid { st.Update(2*node, l, mid, idx, val) } else
                  { st.Update(2*node+1, mid+1, r, idx, val) }
    st.tree[node] = st.tree[2*node] + st.tree[2*node+1]
}

func main() {
    arr := []int{1, 3, 5, 7, 9, 11}
    st := Build(arr)
    fmt.Println(st.Query(1, 0, 5, 1, 4))  // 24
    st.Update(1, 0, 5, 2, 10)
    fmt.Println(st.Query(1, 0, 5, 0, 5))  // 46
}
